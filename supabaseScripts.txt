-- ============================================================
-- STEP 1: Enable Extensions
-- Run from Dashboard: Database → Extensions → enable "vector" and "pgcrypto"
-- Or run here:
-- ============================================================
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pgcrypto;


-- ============================================================
-- STEP 2: Create the documents table
-- NOTE: embedding dimension 768 matches nomic-embed-text.
-- If you change the embeddings model, update vector(768) here
-- AND in the match_documents function below.
-- ============================================================
CREATE TABLE documents (
  id bigserial PRIMARY KEY,
  content text,          -- corresponds to Document.pageContent
  metadata jsonb,        -- corresponds to Document.metadata
  embedding vector(768)
);


-- ============================================================
-- STEP 3: Create the match_documents search function
-- ============================================================
CREATE FUNCTION match_documents (
  query_embedding vector(768),
  match_count int DEFAULT null,
  filter jsonb DEFAULT '{}'
) RETURNS TABLE (
  id bigint,
  content text,
  metadata jsonb,
  embedding jsonb,
  similarity float
)
LANGUAGE plpgsql
AS $$
#variable_conflict use_column
BEGIN
  RETURN QUERY
  SELECT
    id,
    content,
    metadata,
    (embedding::text)::jsonb AS embedding,
    1 - (documents.embedding <=> query_embedding) AS similarity
  FROM documents
  WHERE metadata @> filter
  ORDER BY documents.embedding <=> query_embedding
  LIMIT match_count;
END;
$$;


-- ============================================================
-- STEP 4: Create the users table
-- ============================================================
CREATE TABLE users (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  firstName text,
  lastName text,
  username text UNIQUE,
  password varchar
);


-- ============================================================
-- STEP 5: Create the password hashing trigger
-- Hashes passwords with bcrypt via pgcrypto before insert/update.
-- Do NOT hash passwords in application code — this trigger handles it.
-- ============================================================
CREATE OR REPLACE FUNCTION hash_password_trigger()
RETURNS TRIGGER AS $$
BEGIN
    NEW.password = crypt(NEW.password, gen_salt('bf'));
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER before_insert_or_update
BEFORE INSERT OR UPDATE ON users
FOR EACH ROW
EXECUTE FUNCTION hash_password_trigger();


-- ============================================================
-- STEP 6: Create the verify_password RPC
-- ============================================================
CREATE OR REPLACE FUNCTION verify_password (
  input_username TEXT,
  input_password TEXT
) RETURNS RECORD AS $$
DECLARE
    user_record RECORD;
BEGIN
    SELECT u.firstname, u.lastname, u.username
    INTO user_record
    FROM users u
    WHERE u.username = input_username
    AND u.password = crypt(input_password, u.password);

    IF user_record IS NULL THEN
        RAISE EXCEPTION 'Invalid username or password';
    END IF;

    RETURN user_record;
END;
$$ LANGUAGE plpgsql;


-- ============================================================
-- STEP 7: Create the delete_documents_by_user RPC
-- ============================================================
CREATE OR REPLACE FUNCTION delete_documents_by_user(userid TEXT)
RETURNS VOID AS $$
BEGIN
    DELETE FROM documents
    WHERE metadata @> FORMAT('{"userId": "%s"}', userid)::JSONB;
END;
$$ LANGUAGE plpgsql;

-- ============================================================
-- STEP 9: Hybrid search (keyword + vector, reciprocal rank fusion)
-- Adds a generated full-text column to documents (auto-populated,
-- no ingestion changes needed) and the hybrid_search RPC used by
-- the "hybrid" RAG mode. Safe to run on an existing project.
-- NOTE: vector(768) must stay in sync with match_documents above.
-- ============================================================
ALTER TABLE documents
  ADD COLUMN fts tsvector GENERATED ALWAYS AS (to_tsvector('english', content)) STORED;

CREATE INDEX documents_fts_idx ON documents USING gin(fts);

CREATE OR REPLACE FUNCTION hybrid_search(
  query_text text,
  query_embedding vector(768),
  match_count int DEFAULT 4,
  filter jsonb DEFAULT '{}',
  full_text_weight float DEFAULT 1,
  semantic_weight float DEFAULT 1,
  rrf_k int DEFAULT 50
) RETURNS TABLE (
  id bigint,
  content text,
  metadata jsonb
)
LANGUAGE sql
AS $$
WITH full_text AS (
  SELECT documents.id,
         row_number() OVER (
           ORDER BY ts_rank_cd(documents.fts, websearch_to_tsquery('english', query_text)) DESC
         ) AS rank_ix
  FROM documents
  WHERE documents.fts @@ websearch_to_tsquery('english', query_text)
    AND documents.metadata @> filter
  ORDER BY rank_ix
  LIMIT match_count * 2
),
semantic AS (
  SELECT documents.id,
         row_number() OVER (
           ORDER BY documents.embedding <=> query_embedding
         ) AS rank_ix
  FROM documents
  WHERE documents.metadata @> filter
  ORDER BY rank_ix
  LIMIT match_count * 2
)
SELECT d.id, d.content, d.metadata
FROM full_text
FULL OUTER JOIN semantic ON full_text.id = semantic.id
JOIN documents d ON d.id = COALESCE(full_text.id, semantic.id)
ORDER BY
  COALESCE(1.0 / (rrf_k + full_text.rank_ix), 0.0) * full_text_weight +
  COALESCE(1.0 / (rrf_k + semantic.rank_ix), 0.0) * semantic_weight
  DESC
LIMIT match_count;
$$;


-- ============================================================
-- STEP 10: Sentence-window retrieval (sentence-level table + RPCs)
-- Mirrors documents/match_documents/delete_documents_by_user exactly, but as its own
-- table — every other mode's retriever filters only by {userId}, and Postgres jsonb
-- containment (`@>`) is a superset match, so if sentence-level rows lived in the
-- `documents` table they'd silently match that same {userId} filter and leak into
-- every other mode's retrieved context. A separate table keeps the two granularities
-- from ever mixing. Used by the "sentence-window" RAG mode — without this step, that
-- mode errors at query time (the RPC doesn't exist). Safe to run on an existing project.
-- NOTE: vector(768) must stay in sync with match_documents above.
-- ============================================================
CREATE TABLE sentence_documents (
  id bigserial PRIMARY KEY,
  content text,          -- the bare sentence (what gets embedded)
  metadata jsonb,         -- {documentName, pageNumber, userId, sentenceIndex, windowText}
  embedding vector(768)
);

CREATE FUNCTION match_sentence_documents (
  query_embedding vector(768),
  match_count int DEFAULT null,
  filter jsonb DEFAULT '{}'
) RETURNS TABLE (
  id bigint,
  content text,
  metadata jsonb,
  embedding jsonb,
  similarity float
)
LANGUAGE plpgsql
AS $$
#variable_conflict use_column
BEGIN
  RETURN QUERY
  SELECT
    id,
    content,
    metadata,
    (embedding::text)::jsonb AS embedding,
    1 - (sentence_documents.embedding <=> query_embedding) AS similarity
  FROM sentence_documents
  WHERE metadata @> filter
  ORDER BY sentence_documents.embedding <=> query_embedding
  LIMIT match_count;
END;
$$;

CREATE OR REPLACE FUNCTION delete_sentence_documents_by_user(userid TEXT)
RETURNS VOID AS $$
BEGIN
    DELETE FROM sentence_documents
    WHERE metadata @> FORMAT('{"userId": "%s"}', userid)::JSONB;
END;
$$ LANGUAGE plpgsql;


-- ============================================================
-- STEP 11: Parent-document retrieval (child-chunk table + RPCs)
-- Mirrors documents/match_documents/delete_documents_by_user exactly, but as its own

-- — without this step, that mode errors at query time. Safe to run on an existing project.
-- NOTE: vector(768) must stay in sync with match_documents above.
-- ============================================================

-- ============================================================
-- STEP 12: Graph RAG (entity/relation graph + traversal)
-- Adds a small knowledge graph extracted at ingestion time: entities mentioned in each
-- parent chunk, which chunk(s) mention each entity, and relations between entity pairs.
-- Unlike the other STEPs above, these are plain typed columns (not jsonb metadata) — there's
-- no embedding column here (retrieval matches entities by name, not vector similarity) and
-- traversal needs real relational joins/foreign keys, which plain columns suit better than
-- jsonb containment filtering. chunk_id references `documents.id` directly. Used by the
-- "graph" RAG mode — without this step, that mode errors at query time (same as hybrid's
-- STEP 9, sentence-window's STEP 10, and parent-document's STEP 11 dependencies). Safe to
-- run on an existing project.
-- ============================================================
CREATE TABLE graph_entities (
  id bigserial PRIMARY KEY,
  name text NOT NULL,      -- normalized (trimmed, lowercased) entity name
  type text,                -- free-form entity type from extraction, e.g. "person" (nullable)
  user_id text NOT NULL
);

CREATE INDEX graph_entities_user_id_name_idx ON graph_entities (user_id, name);

CREATE TABLE graph_entity_mentions (
  entity_id bigint NOT NULL REFERENCES graph_entities(id) ON DELETE CASCADE,
  chunk_id bigint NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
  PRIMARY KEY (entity_id, chunk_id)
);

CREATE TABLE graph_relations (
  id bigserial PRIMARY KEY,
  source_entity_id bigint NOT NULL REFERENCES graph_entities(id) ON DELETE CASCADE,
  target_entity_id bigint NOT NULL REFERENCES graph_entities(id) ON DELETE CASCADE,
  label text,
  user_id text NOT NULL
);

CREATE INDEX graph_relations_user_id_idx ON graph_relations (user_id);

CREATE OR REPLACE FUNCTION delete_graph_data_by_user(userid TEXT)
RETURNS VOID AS $$
BEGIN
    DELETE FROM graph_relations WHERE user_id = userid;
    DELETE FROM graph_entity_mentions
      WHERE entity_id IN (SELECT id FROM graph_entities WHERE user_id = userid);
    DELETE FROM graph_entities WHERE user_id = userid;
END;
$$ LANGUAGE plpgsql;
